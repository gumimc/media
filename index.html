const ALLOWED_ORIGIN = "https://gumimc.github.io"; function corsHeaders(extra = {}) { return { "Access-Control-Allow-Origin": ALLOWED_ORIGIN, "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, X-Object-Key, X-File-Name", ...extra, }; } function resolveB2Config(env) { const keyId = (env.B2_KEY_ID || env.KEY_ID || "").trim(); const appKey = ( env.B2_APPLICATION_KEY || env.B2_APP_KEY || env.APPLICATION_KEY || "" ).trim(); const bucketName = ( env.B2_BUCKET_NAME || env.B2_BUCKET || "gumimc-media-files" ).trim(); if (!keyId || !appKey) { throw new Error("Worker 환경변수 B2_KEY_ID 또는 B2_APPLICATION_KEY가 없습니다."); } return { keyId, appKey, bucketName }; } async function authorizeB2Account(keyId, appKey) { const basicAuth = btoa(`${keyId}:${appKey}`); const res = await fetch("https://api.backblazeb2.com/b2api/v2/b2_authorize_account", { method: "GET", headers: { Authorization: `Basic ${basicAuth}` }, }); const data = await res.json(); if (!res.ok || !data.authorizationToken) { throw new Error(`B2 인증 실패: ${data.message || res.status}`); } return data; } async function getBucketId(authData, bucketName) { const res = await fetch(`${authData.apiUrl}/b2api/v2/b2_list_buckets`, { method: "POST", headers: { Authorization: authData.authorizationToken, "Content-Type": "application/json", }, body: JSON.stringify({ accountId: authData.accountId, bucketName, }), }); const data = await res.json(); const bucket = (data.buckets || []).find((b) => b.bucketName === bucketName); if (!res.ok || !bucket) { throw new Error(`버킷(${bucketName})을 찾을 수 없습니다.`); } return bucket.bucketId; } async function sha1Hex(arrayBuffer) { const hash = await crypto.subtle.digest("SHA-1", arrayBuffer); return [...new Uint8Array(hash)] .map((b) => b.toString(16).padStart(2, "0")) .join(""); } function encodeB2FileName(objectKey) { return objectKey .split("/") .filter(Boolean) .map((seg) => encodeURIComponent(seg)) .join("/"); } async function handleListFiles(env) { const { keyId, appKey, bucketName } = resolveB2Config(env); const authData = await authorizeB2Account(keyId, appKey); const bucketId = await getBucketId(authData, bucketName); const files = []; let startFileName = null; while (true) { const body = { bucketId, maxFileCount: 1000 }; if (startFileName) body.startFileName = startFileName; const res = await fetch(`${authData.apiUrl}/b2api/v2/b2_list_file_names`, { method: "POST", headers: { Authorization: authData.authorizationToken, "Content-Type": "application/json", }, body: JSON.stringify(body), }); const data = await res.json(); if (!res.ok) { throw new Error(`B2 파일 목록 조회 실패: ${data.message || res.status}`); } for (const f of data.files || []) { const parts = String(f.fileName || "").split("/"); const deptName = parts[1] || "기타"; const kind = parts[2] || "file"; const rawName = parts.slice(3).join("/") || f.fileName; const cleanName = rawName.replace(/^f_\d+_[a-z0-9]+_/i, ""); files.push({ objectKey: f.fileName, deptName, kind, fileName: cleanName || rawName, size: Number(f.contentLength || 0), uploadTimestamp: Number(f.uploadTimestamp || 0), }); } if (!data.nextFileName) break; startFileName = data.nextFileName; } return new Response(JSON.stringify({ ok: true, files }), { status: 200, headers: corsHeaders({ "Content-Type": "application/json" }), }); } async function handleUpload(request, env, url) { const { keyId, appKey, bucketName } = resolveB2Config(env); const rawKey = url.searchParams.get("objectKey") || request.headers.get("X-Object-Key") || ""; const objectKey = decodeURIComponent(rawKey).trim(); if (!objectKey) { return new Response(JSON.stringify({ error: "objectKey가 필요합니다." }), { status: 400, headers: corsHeaders({ "Content-Type": "application/json" }), }); } const fileBuffer = await request.arrayBuffer(); const sha1 = await sha1Hex(fileBuffer); const contentType = request.headers.get("Content-Type") || "b2/x-auto"; const authData = await authorizeB2Account(keyId, appKey); const bucketId = await getBucketId(authData, bucketName); const upUrlRes = await fetch(`${authData.apiUrl}/b2api/v2/b2_get_upload_url`, { method: "POST", headers: { Authorization: authData.authorizationToken, "Content-Type": "application/json", }, body: JSON.stringify({ bucketId }), }); const upUrlData = await upUrlRes.json(); if (!upUrlRes.ok || !upUrlData.uploadUrl) { throw new Error(`B2 uploadUrl 발급 실패: ${upUrlData.message || upUrlRes.status}`); } const uploadRes = await fetch(upUrlData.uploadUrl, { method: "POST", headers: { Authorization: upUrlData.authorizationToken, "X-Bz-File-Name": encodeB2FileName(objectKey), "Content-Type": contentType, "Content-Length": String(fileBuffer.byteLength), "X-Bz-Content-Sha1": sha1, }, body: fileBuffer, }); const uploadJson = await uploadRes.json().catch(() => ({})); if (!uploadRes.ok) { throw new Error(`B2 파일 업로드 실패: ${uploadJson.message || uploadRes.status}`); } const downloadUrl = `${url.origin}${url.pathname}?objectKey=${encodeURIComponent(objectKey)}`; return new Response( JSON.stringify({ ok: true, objectKey, fileId: uploadJson.fileId || "", downloadUrl, }), { status: 200, headers: corsHeaders({ "Content-Type": "application/json" }), } ); } async function handleDownload(request, env, url) { const { keyId, appKey, bucketName } = resolveB2Config(env); const objectKey = (url.searchParams.get("objectKey") || "").trim(); const fileName = (url.searchParams.get("fileName") || "").trim(); if (!objectKey) { return new Response(JSON.stringify({ error: "objectKey가 필요합니다." }), { status: 400, headers: corsHeaders({ "Content-Type": "application/json" }), }); } const authData = await authorizeB2Account(keyId, appKey); const dlUrl = `${authData.downloadUrl}/file/${encodeURIComponent(bucketName)}/${encodeB2FileName(objectKey)}`; const b2Res = await fetch(dlUrl, { method: "GET", headers: { Authorization: authData.authorizationToken, }, }); if (!b2Res.ok) { const errText = await b2Res.text().catch(() => ""); return new Response(errText || `다운로드 실패 (${b2Res.status})`, { status: b2Res.status, headers: corsHeaders({ "Content-Type": "text/plain; charset=utf-8" }), }); } const headers = corsHeaders({ "Content-Type": b2Res.headers.get("Content-Type") || "application/octet-stream", }); if (fileName) { const safeAscii = fileName.replace(/[^\x20-\x7E]/g, "_"); headers["Content-Disposition"] = `attachment; filename="${safeAscii}"; filename*=UTF-8''${encodeURIComponent(fileName)}`; } return new Response(b2Res.body, { status: 200, headers, }); } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders() }); } const url = new URL(request.url); try { if (request.method === "POST") { return await handleUpload(request, env, url); } if (request.method === "GET") { if (url.searchParams.get("action") === "list") { return await handleListFiles(env); } return await handleDownload(request, env, url); } return new Response(JSON.stringify({ error: "지원하지 않는 메서드" }), { status: 405, headers: corsHeaders({ "Content-Type": "application/json" }), }); } catch (err) { return new Response( JSON.stringify({ error: err.message || "워커 처리 오류" }), { status: 500, headers: corsHeaders({ "Content-Type": "application/json" }), } ); } }, };
